import { useState } from "react";

export default function App() {
  const [task, setTask] = useState("");
  const [tasks, setTasks] = useState([]);

  function addTask() {
    const value = task.trim();
    if (!value) return;
    setTasks([...tasks, { id: Date.now(), text: value, done: false }]);
    setTask("");
  }

  function toggleTask(id) {
    setTasks(tasks.map(item =>
      item.id === id ? { ...item, done: !item.done } : item
    ));
  }

  function deleteTask(id) {
    setTasks(tasks.filter(item => item.id !== id));
  }

  return (
    <main style={styles.page}>
      <h1>To-Do List</h1>
      <form onSubmit={e => { e.preventDefault(); addTask(); }} style={styles.row}>
        <input style={styles.input} value={task} onChange={e => setTask(e.target.value)} placeholder="Enter a task" />
        <button style={styles.button}>Add</button>
      </form>
      <p>{tasks.filter(t => !t.done).length} task(s) remaining</p>
      {tasks.length === 0 && <p>No tasks yet. Add your first task.</p>}
      {tasks.map(item => (
        <div key={item.id} style={styles.item}>
          <label style={{ textDecoration: item.done ? "line-through" : "none", flex: 1 }}>
            <input type="checkbox" checked={item.done} onChange={() => toggleTask(item.id)} /> {item.text}
          </label>
          <button onClick={() => deleteTask(item.id)}>Delete</button>
        </div>
      ))}
    </main>
  );
}

const styles = {
  page: { maxWidth: 520, margin: "40px auto", padding: 24, fontFamily: "Arial", background: "#f4f7fb", borderRadius: 12 },
  row: { display: "flex", gap: 8 },
  input: { flex: 1, padding: 12, border: "1px solid #aaa", borderRadius: 6 },
  button: { padding: "10px 18px", cursor: "pointer" },
  item: { display: "flex", alignItems: "center", gap: 10, padding: 10, marginTop: 8, background: "white", borderRadius: 6 }
};

import { useState } from "react";

export default function App() {
  const [first, setFirst] = useState("");
  const [second, setSecond] = useState("");
  const [operator, setOperator] = useState("+");
  const [result, setResult] = useState(null);
  const [error, setError] = useState("");

  function calculate() {
    const a = Number(first);
    const b = Number(second);
    if (first.trim() === "" || second.trim() === "") {
      setError("Enter both numbers.");
      setResult(null);
      return;
    }
    let answer;
    if (operator === "+") answer = a + b;
    else if (operator === "-") answer = a - b;
    else if (operator === "*") answer = a * b;
    else {
      if (b === 0) {
        setError("Cannot divide by zero.");
        setResult(null);
        return;
      }
      answer = a / b;
    }
    setError("");
    setResult(answer);
  }

  return (
    <main style={s.page}>
      <h1>Calculator</h1>
      <input style={s.input} type="number" value={first} onChange={e => setFirst(e.target.value)} placeholder="First number" />
      <select style={s.input} value={operator} onChange={e => setOperator(e.target.value)}>
        <option value="+">Addition (+)</option><option value="-">Subtraction (-)</option>
        <option value="*">Multiplication (*)</option><option value="/">Division (/)</option>
      </select>
      <input style={s.input} type="number" value={second} onChange={e => setSecond(e.target.value)} placeholder="Second number" />
      <button style={s.button} onClick={calculate}>Calculate</button>
      {error && <p style={{ color: "red" }}>{error}</p>}
      {result !== null && <h2>Result: {result}</h2>}
      <button onClick={() => { setFirst(""); setSecond(""); setResult(null); setError(""); }}>Clear</button>
    </main>
  );
}
const s = { page: { maxWidth: 400, margin: "40px auto", padding: 24, fontFamily: "Arial", border: "1px solid #ddd", borderRadius: 10 }, input: { display: "block", width: "100%", boxSizing: "border-box", padding: 10, margin: "12px 0" }, button: { padding: 10, marginRight: 8, cursor: "pointer" } };

import { useState } from "react";

const questions = [
  { question: "Which library is used to build user interfaces?", options: ["React", "MongoDB", "Node only", "SQL"], answer: "React" },
  { question: "Which hook stores component state?", options: ["useRoute", "useState", "usePage", "useHTML"], answer: "useState" },
  { question: "Which syntax is commonly used to write HTML-like UI in React?", options: ["JSX", "SQL", "XML only", "Markdown"], answer: "JSX" },
  { question: "Which prop helps React identify list items?", options: ["idName", "key", "class", "indexOnly"], answer: "key" }
];

export default function App() {
  const [index, setIndex] = useState(0);
  const [selected, setSelected] = useState("");
  const [score, setScore] = useState(0);
  const [finished, setFinished] = useState(false);

  function nextQuestion() {
    if (!selected) return;
    const newScore = score + (selected === questions[index].answer ? 1 : 0);
    setScore(newScore);
    if (index === questions.length - 1) setFinished(true);
    else setIndex(index + 1);
    setSelected("");
  }

  function restart() {
    setIndex(0); setSelected(""); setScore(0); setFinished(false);
  }

  return (
    <main style={s.card}>
      <h1>React Quiz</h1>
      {finished ? (
        <>
          <h2>Quiz Completed!</h2>
          <p>Your score: {score} / {questions.length}</p>
          <button onClick={restart}>Try Again</button>
        </>
      ) : (
        <>
          <p>Question {index + 1} of {questions.length}</p>
          <h2>{questions[index].question}</h2>
          {questions[index].options.map(option => (
            <label key={option} style={s.option}>
              <input type="radio" name="answer" value={option} checked={selected === option} onChange={() => setSelected(option)} />
              {option}
            </label>
          ))}
          <button disabled={!selected} onClick={nextQuestion}>{index === questions.length - 1 ? "Finish" : "Next"}</button>
        </>
      )}
    </main>
  );
}
const s = { card: { maxWidth: 520, margin: "40px auto", padding: 24, fontFamily: "Arial", border: "1px solid #ddd", borderRadius: 12 }, option: { display: "block", padding: 10, margin: "8px 0", background: "#f3f5f8", borderRadius: 6 } };

import { useState } from "react";

export default function App() {
  const [temperature, setTemperature] = useState("");
  const [from, setFrom] = useState("C");
  const [to, setTo] = useState("F");

  function convert() {
    if (temperature.trim() === "") return "";
    const value = Number(temperature);
    let celsius;
    if (from === "C") celsius = value;
    else if (from === "F") celsius = (value - 32) * 5 / 9;
    else celsius = value - 273.15;

    if (to === "C") return celsius.toFixed(2);
    if (to === "F") return (celsius * 9 / 5 + 32).toFixed(2);
    return (celsius + 273.15).toFixed(2);
  }

  return (
    <main style={s.card}>
      <h1>Temperature Converter</h1>
      <input style={s.input} type="number" value={temperature} onChange={e => setTemperature(e.target.value)} placeholder="Enter temperature" />
      <label>From: <select value={from} onChange={e => setFrom(e.target.value)}><option value="C">Celsius</option><option value="F">Fahrenheit</option><option value="K">Kelvin</option></select></label>
      <br /><br />
      <label>To: <select value={to} onChange={e => setTo(e.target.value)}><option value="C">Celsius</option><option value="F">Fahrenheit</option><option value="K">Kelvin</option></select></label>
      <h2>Converted value: {temperature.trim() === "" ? "—" : convert()} {to}</h2>
    </main>
  );
}
const s = { card: { maxWidth: 450, margin: "40px auto", padding: 24, fontFamily: "Arial", border: "1px solid #ddd", borderRadius: 10 }, input: { display: "block", width: "100%", padding: 10, boxSizing: "border-box", margin: "14px 0" } };

import { useState } from "react";

export default function App() {
  const [description, setDescription] = useState("");
  const [amount, setAmount] = useState("");
  const [type, setType] = useState("expense");

        <input style={s.input} value={description} onChange={e => setDescription(e.target.value)} placeholder="Description" />
        <input style={s.input} type="number" min="0.01" step="0.01" value={amount} onChange={e => setAmount(e.target.value)} placeholder="Amount" />

      <h2>History</h2>

    </main>

import { useState } from "react";

export default function App() {
  const [weight, setWeight] = useState("");
  const [height, setHeight] = useState("");
  const [bmi, setBmi] = useState(null);
  const [message, setMessage] = useState("");

  function calculateBMI(e) {
    e.preventDefault();
    const w = Number(weight);
    const h = Number(height) / 100;
    if (w <= 0 || h <= 0 || !weight || !height) {
      setMessage("Enter a valid weight and height.");
      setBmi(null);
      return;
    }
    const value = w / (h * h);
    setBmi(value.toFixed(2));
    if (value < 18.5) setMessage("Underweight range");
    else if (value < 25) setMessage("Normal range");
    else if (value < 30) setMessage("Overweight range");
    else setMessage("Obesity range");
  }

  return (
    <main style={s.card}>
      <h1>BMI Calculator</h1>
      <form onSubmit={calculateBMI}>
        <label>Weight (kg)<input style={s.input} type="number" min="1" step="any" value={weight} onChange={e => setWeight(e.target.value)} /></label>
        <label>Height (cm)<input style={s.input} type="number" min="1" step="any" value={height} onChange={e => setHeight(e.target.value)} /></label>
        <button>Calculate BMI</button>
      </form>
      {message && <p>{message}</p>}
      {bmi !== null && <h2>Your BMI: {bmi}</h2>}
    </main>
  );
}
const s = { card: { maxWidth: 420, margin: "40px auto", padding: 24, fontFamily: "Arial", border: "1px solid #ddd", borderRadius: 10 }, input: { display: "block", width: "100%", padding: 10, boxSizing: "border-box", margin: "8px 0 16px" } };

import { useState } from "react";

export default function App() {
  const [word, setWord] = useState("");
  const [definition, setDefinition] = useState(null);
  const [loading, setLoading] = useState(false);
  const [error, setError] = useState("");

  async function searchWord(e) {
    e.preventDefault();
    if (!word.trim()) return;
    setLoading(true); setError(""); setDefinition(null);
    try {
      const response = await fetch("https://api.dictionaryapi.dev/api/v2/entries/en/" + encodeURIComponent(word.trim()));
      if (!response.ok) throw new Error("Word not found.");
      const data = await response.json();
      const meaning = data[0].meanings?.[0];
      const def = meaning?.definitions?.[0]?.definition;
      setDefinition({ word: data[0].word, part: meaning?.partOfSpeech || "", text: def || "No definition available." });
    } catch {
      setError("Definition not found. Check spelling or internet connection.");
    } finally {
      setLoading(false);
    }
  }

  return (
    <main style={s.card}>
      <h1>Dictionary</h1>
      <form onSubmit={searchWord} style={s.row}>
        <input style={s.input} value={word} onChange={e => setWord(e.target.value)} placeholder="Enter an English word" />
        <button>Search</button>
      </form>
      {loading && <p>Searching...</p>}
      {error && <p style={{ color: "red" }}>{error}</p>}
      {definition && <article><h2>{definition.word}</h2><p><em>{definition.part}</em></p><p>{definition.text}</p></article>}
    </main>
  );
}
const s = { card: { maxWidth: 600, margin: "40px auto", padding: 24, fontFamily: "Arial", border: "1px solid #ddd", borderRadius: 10 }, row: { display: "flex", gap: 8 }, input: { flex: 1, minWidth: 0, padding: 10 } };

import { useState } from "react";

export default function App() {
  const [query, setQuery] = useState("");

  const [error, setError] = useState("");

  async function searchBooks(e) {
    e.preventDefault();
    if (!query.trim()) return;
    setLoading(true); setError(""); setBooks([]);
    try {
      const response = await fetch("https://www.googleapis.com/books/v1/volumes?q=" + encodeURIComponent(query.trim()) + "&maxResults=12");
      if (!response.ok) throw new Error("Request failed");
      const data = await response.json();
      setBooks(data.items || []);
      if (!data.items?.length) setError("No books found.");
    } catch {
      setError("Could not load books. Check your internet connection.");
    } finally {
      setLoading(false);
    }

import { useState } from "react";

export default function App() {
  const [birthDate, setBirthDate] = useState("");
  const [age, setAge] = useState(null);
  const [error, setError] = useState("");

  function calculateAge(e) {
    e.preventDefault();
    if (!birthDate) { setError("Select your date of birth."); setAge(null); return; }
    const birth = new Date(birthDate + "T00:00:00");
    const today = new Date();
    if (birth > today) { setError("Birth date cannot be in the future."); setAge(null); return; }
    let years = today.getFullYear() - birth.getFullYear();
    let months = today.getMonth() - birth.getMonth();
    let days = today.getDate() - birth.getDate();
    if (days < 0) {
      months--;
      const previousMonthDays = new Date(today.getFullYear(), today.getMonth(), 0).getDate();
      days += previousMonthDays;
    }
    if (months < 0) { years--; months += 12; }
    setAge({ years, months, days });
    setError("");
  }

  return (
    <main style={s.card}>
      <h1>Age Calculator</h1>
      <form onSubmit={calculateAge}>
        <label>Date of birth<input style={s.input} type="date" max={new Date().toISOString().split("T")[0]} value={birthDate} onChange={e => setBirthDate(e.target.value)} /></label>
        <button>Calculate Age</button>
      </form>
      {error && <p style={{ color: "red" }}>{error}</p>}
      {age && <h2>{age.years} years, {age.months} months, {age.days} days</h2>}
    </main>
  );
}
const s = { card: { maxWidth: 450, margin: "40px auto", padding: 24, fontFamily: "Arial", border: "1px solid #ddd", borderRadius: 10 }, input: { display: "block", width: "100%", boxSizing: "border-box", padding: 10, margin: "10px 0 16px" } };

import { useState } from "react";

const initialItems = [
  { id: 1, name: "Notebook", price: 50, quantity: 1 },
  { id: 2, name: "Pen", price: 10, quantity: 2 }
];

export default function App() {
  const [items, setItems] = useState(initialItems);
  const [name, setName] = useState("");
  const [price, setPrice] = useState("");
  const [quantity, setQuantity] = useState("1");
  const TAX_RATE = 0.05;

  function addItem(e) {
    e.preventDefault();
    if (!name.trim() || price === "" || Number(price) < 0 || Number(quantity) < 1) return;
    setItems([...items, { id: Date.now(), name: name.trim(), price: Number(price), quantity: Number(quantity) }]);
    setName(""); setPrice(""); setQuantity("1");
  }

  const subtotal = items.reduce((sum, item) => sum + item.price * item.quantity, 0);
  const tax = subtotal * TAX_RATE;
  const total = subtotal + tax;

  return (
    <main style={s.page}>
      <h1>Shopping Bill Calculator</h1>
      <form onSubmit={addItem} style={s.form}>
        <input style={s.input} value={name} onChange={e => setName(e.target.value)} placeholder="Item name" />
        <input style={s.input} type="number" min="0" step="0.01" value={price} onChange={e => setPrice(e.target.value)} placeholder="Price" />
        <input style={s.input} type="number" min="1" value={quantity} onChange={e => setQuantity(e.target.value)} placeholder="Qty" />

      </form>

          <td style={s.cell}>{item.quantity}</td><td style={s.cell}>₹{(item.price * item.quantity).toFixed(2)}</td>

        </tr>)}</tbody>
      </table>
      <section style={s.total}><p>Subtotal: ₹{subtotal.toFixed(2)}</p><p>Tax (5%): ₹{tax.toFixed(2)}</p><h2>Grand Total: ₹{total.toFixed(2)}</h2></section>
      <button onClick={() => window.print()}>Print Bill</button>
    </main>
  );
}
